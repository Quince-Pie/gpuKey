/* Independent dependency probe: no gpuKey code, no input and no drain. */
#include <libavcodec/avcodec.h>
#include <libavutil/hwcontext.h>
#include <stdio.h>
#include <stdlib.h>

#define CHECK(x) do { if (!(x)) { fprintf(stderr, "failed at %d: %s\n", __LINE__, #x); exit(2); } } while (0)

int main(void)
{
    for (unsigned i = 0; i < 4; ++i) {
        AVBufferRef *device = nullptr;
        CHECK(av_hwdevice_ctx_create(&device, AV_HWDEVICE_TYPE_VULKAN, nullptr, nullptr, 0) >= 0);
        AVBufferRef *frames = av_hwframe_ctx_alloc(device);
        CHECK(frames);
        AVHWFramesContext *fc = (void *)frames->data;
        fc->format = AV_PIX_FMT_VULKAN;
        fc->sw_format = AV_PIX_FMT_NV12;
        fc->width = 320;
        fc->height = 240;
        fc->initial_pool_size = 4;
        CHECK(av_hwframe_ctx_init(frames) >= 0);
        const AVCodec *codec = avcodec_find_encoder_by_name("av1_vulkan");
        CHECK(codec);
        AVCodecContext *context = avcodec_alloc_context3(codec);
        CHECK(context);
        context->hw_frames_ctx = av_buffer_ref(frames);
        CHECK(context->hw_frames_ctx);
        context->width = 320;
        context->height = 240;
        context->pix_fmt = AV_PIX_FMT_VULKAN;
        context->sw_pix_fmt = AV_PIX_FMT_NV12;
        context->time_base = (AVRational){1, 1000000};
        context->framerate = (AVRational){60, 1};
        context->max_b_frames = 0;
        context->gop_size = 120;
        context->flags |= AV_CODEC_FLAG_GLOBAL_HEADER;
        AVDictionary *options = nullptr;
        av_dict_set(&options, "rc_mode", "cqp", 0);
        av_dict_set(&options, "qp", "32", 0);
        av_dict_set(&options, "async_depth", "2", 0);
        CHECK(avcodec_open2(context, codec, &options) >= 0);
        CHECK(!av_dict_count(options));
        av_dict_free(&options);
        avcodec_free_context(&context);
        av_buffer_unref(&frames);
        av_buffer_unref(&device);
    }
    puts("four direct FFmpeg Vulkan open/free cycles completed");
}
